\section{Algorithms}
\label{sec:appendix-algorithms}

We summarize the time steps of the constrained (\S\ref{sec:flow-constrained}) and penalty-based (\S\ref{sec:flow-penalty}) integrators, together with the Newton solves they rely on.
The Newton solves of $\jointdual$ and $\jointdual_\alpha$ (Algorithms~\ref{alg:flow-constrained-newton} and~\ref{alg:newton}) take Levenberg--Marquardt steps (Algorithm~\ref{alg:lm-step}), while the solve of $\dualenergy$ (Algorithm~\ref{alg:volume}) backtracks the Newton step via the line search of Algorithm~\ref{alg:line-search}.
Both rebuild the power diagram at every trial step.
In all of them, the negated Hessian is assembled and regularized by Algorithm~\ref{alg:hessian-reg}, as described in \S\ref{sec:results-implementation}, and the solves stop under the test \eqref{eq:results-termination}.

\begin{algorithm}[H]
  \caption{Time step of the constrained integrator (\S\ref{sec:flow-constrained}).}
  \label{alg:flow-constrained}
  \begin{algorithmic}[1]
    \Require \algwrap{particles $\qq^k$, velocities $\vel^k$, generators $\dual^k$ and shifted weights $\combweight^k$ of the power diagram with centroids $\qq^k$, volumes $\pvol$, time step $\tstep$}
    \State $\pred^{k+1} \gets (1 + \tstep^2)\,\qq^k + \tstep\,\vel^k$ \Comment{\eqref{eq:flow-shake-step-pred}}
    \State $\weight^0 \gets \operatorname{arg\,max}_{\weight} \dualenergy(\weight)$ for the sites $\qq^k$ \Comment{\eqref{eq:volume-dual}}
    \Statex \hspace{\algorithmicindent} by Alg.~\ref{alg:volume}, warm-started from $\combweight^k$
    \State $(\shift, \weight, s, \translation) \gets$ Alg.~\ref{alg:flow-constrained-newton} with $\qq^k$, $\pred^{k+1}$, $\dual^k$, $\weight^0$
    \State $\qq^{k+1} \gets \pred^{k+1} - s\,\dual^k - \translation$ \Comment{\eqref{eq:flow-shake-step-corr}}
    \State $\vel^{k+1} \gets (\qq^{k+1} - \qq^k) / \tstep$
    \State $\dual^{k+1} \gets \qq^k + \shift$ \Comment{\eqref{eq:shifted-site-pos}}
    \State $\combweight^{k+1} \gets \weight + \norm{\shift}^2 + 2\,\shift\tr (\qq^k - \pred^{k+1})$ \Comment{\eqref{eq:shifted-site-weight}}
    \State \Return \algwrap{particles $\qq^{k+1}$, velocities $\vel^{k+1}$, generators $\dual^{k+1}$ and shifted weights $\combweight^{k+1}$}
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{Bordered Newton solve of \eqref{eq:flow-line-constraint} (Appendix~\ref{sec:appendix-bordered}).}
  \label{alg:flow-constrained-newton}
  \begin{algorithmic}[1]
    \Require \algwrap{sites $\qq$, targets $\tcentroid$, direction $\dual^k$, initial weights $\weight^0$, volumes $\pvol$, tolerances $\tau_\weight, \tau_\shift$}
    \State build the constraint matrix $\Cmat = [\mathbf{a}, \mathbf{b}_1, \dots, \mathbf{b}_d]\tr$ \Comment{\eqref{eq:appendix-constraint-matrix}}
    \State $\mathbf{z} = (\shift, \weight) \gets (0, \weight^0)$, $\mu \gets 10^{-3}$
    \State build the power diagram of the sites $\qq$ shifted by $\mathbf{z}$ \Comment{\eqref{eq:shifted-site}}
    \Loop
      \State $\mathbf{g} = (\mathbf{g}^\shift_i, g^\weight_i)_i \gets \nabla\jointdual(\mathbf{z})$ \Comment{\eqref{eq:joint-gradient}}
      \State $r_\shift \gets \max_i \norm{\mathbf{g}^\shift_i} / (2\pvol_i^{(d+1)/d})$ \Comment{shift residual}
      \State $\Nmat \gets$ Alg.~\ref{alg:hessian-reg} at $\mathbf{z}$ with $\sigma = \min(1, r_\shift / r^0_\shift)$ and $\beta$
      \State solve for $\mathbf{z}_0$, $\mathbf{z}_a$, $\mathbf{z}_{b_j}$ \Comment{\eqref{eq:flow-constrained-elimination-solve}}
      \State solve for the gauge $(s, \translation)$ \Comment{\eqref{eq:flow-constrained-elimination-gauge}}
      \State $\mathbf{r} = (\mathbf{r}^\shift_i, r^\weight_i)_i \gets \mathbf{g} - 2 \Cmat\tr (s, \translation)$ \Comment{\eqref{eq:flow-constrained-newton}}
      \If{$\mathbf{r}$ passes the test \eqref{eq:results-termination}} \Comment{in place of $\nabla\jointdual$}
        \State \Return $(\shift, \weight)$ and the gauge $(s, \translation)$
      \EndIf
      \State $(\mathbf{z}, \mu) \gets$ Alg.~\ref{alg:lm-step} with $\mathbf{z}$, $\mathbf{g}$, $\Nmat$, $\mu$, $\Cmat$, and $\alpha = \infty$
    \EndLoop
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{Time step of the penalty-based integrator (\S\ref{sec:flow-penalty}).}
  \label{alg:flow-penalty}
  \begin{algorithmic}[1]
    \Require \algwrap{particles $\qq^k$, velocities $\vel^k$, shifted weights $\combweight^k$ of the previous step, volumes $\pvol$, time step $\tstep$, stiffness $\alpha$}
    \State $\weight^0 \gets \operatorname{arg\,max}_{\weight} \dualenergy(\weight)$ for the sites $\qq^k$ \Comment{\eqref{eq:volume-dual}}
    \Statex \hspace{\algorithmicindent} by Alg.~\ref{alg:volume}, warm-started from $\combweight^k$
    \State $(\shift, \weight) \gets$ Alg.~\ref{alg:newton} with sites and targets $\qq^k$
    \Statex \hspace{\algorithmicindent} and initial weights $\weight^0$
    \State $\vel^{k+1} \gets \vel^k - (1 + 1/\alpha)\,\tstep\,\shift$ \Comment{\eqref{eq:flow-softened-step-vel}}
    \State $\qq^{k+1} \gets \qq^k + \tstep\,\vel^{k+1}$ \Comment{\eqref{eq:flow-softened-step-pos}}
    \State $\combweight^{k+1} \gets \weight + \norm{\shift}^2$ \Comment{\eqref{eq:shifted-site-weight}}
    \State \Return \algwrap{particles $\qq^{k+1}$, velocities $\vel^{k+1}$ and shifted weights $\combweight^{k+1}$}
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{Newton solve of $\jointdual_\alpha$ or of $\jointdual$.}
  \label{alg:newton}
  \begin{algorithmic}[1]
    \Require \algwrap{sites $\qq$, targets $\tcentroid$, initial weights $\weight^0$, volumes $\pvol$, stiffness $\alpha$ (optional), tolerances $\tau_\weight, \tau_\shift$}
    \State $\mathbf{z} = (\shift, \weight) \gets (0, \weight^0)$, $\mu \gets 10^{-3}$
    \State build the power diagram of the sites $\qq$ shifted by $\mathbf{z}$ \Comment{\eqref{eq:shifted-site}}
    \Loop
      \State $\mathbf{g} = (\mathbf{g}^\shift_i, g^\weight_i)_i \gets \nabla\jointdual_\alpha(\mathbf{z})$ \Comment{\eqref{eq:joint-gradient}, \eqref{eq:regularized-gradient-shift}}
      \State $r_\shift \gets \max_i \norm{\mathbf{g}^\shift_i} / (2\pvol_i^{(d+1)/d})$ \Comment{shift residual}
      \If{$\mathbf{g}$ passes the test \eqref{eq:results-termination}}
        \State \Return $(\shift, \weight)$
      \EndIf
      \State $\Nmat \gets$ Alg.~\ref{alg:hessian-reg} at $\mathbf{z}$ with $\sigma = \min(1, r_\shift / r^0_\shift)$ and $\beta$
      \State $(\mathbf{z}, \mu) \gets$ Alg.~\ref{alg:lm-step} with $\mathbf{z}$, $\mathbf{g}$, $\Nmat$, $\mu$, and $\alpha$
    \EndLoop
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{Levenberg--Marquardt step on $\jointdual_\alpha$ or on $\jointdual$.}
  \label{alg:lm-step}
  \begin{algorithmic}[1]
    \Require \algwrap{multipliers $\mathbf{z}$, gradient $\mathbf{g}$, matrix $\Nmat$, damping $\mu$, constraints $\Cmat$ (optional), stiffness $\alpha$}
    \State $\mathrm{D} \gets \mathrm{diag}(\Nmat)$
    \For{$k = 1, 2, \dots$} \Comment{trials}
      \State solve $(\Nmat + \mu\,\mathrm{D})\,\delta\mathbf{z} = \mathbf{g}$, subject to $\Cmat\,\delta\mathbf{z} = 0$ \Comment{\eqref{eq:results-lm-step}, \eqref{eq:flow-constrained-elimination-update}}
      \State $\mathbf{z}' \gets \mathbf{z} + \delta\mathbf{z}$
      \State build the power diagram of the sites $\qq$ shifted by $\mathbf{z}'$ \Comment{\eqref{eq:shifted-site}}
      \State $\varrho \gets \bigl(\jointdual_\alpha(\mathbf{z}') - \jointdual_\alpha(\mathbf{z})\bigr) / \bigl(\mathbf{g}\tr\delta\mathbf{z} - \tfrac{1}{2}\,\delta\mathbf{z}\tr\Nmat\,\delta\mathbf{z}\bigr)$ \Comment{gain ratio}
      \If{no cell is empty and $\varrho > 10^{-4}$}
        \State \Return \algwrap{$\mathbf{z}'$, its power diagram, and $\mu \max\bigl(\tfrac{1}{3}, 1 - (2\varrho - 1)^3\bigr)$}
      \EndIf
      \State $\mu \gets 2^k\,\mu$ \Comment{rejected}
    \EndFor
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{Regularized negated Hessian of $\jointdual_\alpha$, $\jointdual$, or $\dualenergy$ (\S\ref{sec:results-implementation}).}
  \label{alg:hessian-reg}
  \begin{algorithmic}[1]
    \Require \algwrap{power diagram at $\mathbf{z}$, volumes $\pvol$, activity $\sigma \in [0, 1]$, radius ratio $\beta$}
    \State $\bar{r} \gets (\lvert\domain\rvert / n)^{1/d}$, $\hreg \gets 10^{-3}$ \Comment{average cell size}
    \State $\Nmat \gets -\nabla^2\jointdual_\alpha(\mathbf{z})$, with $\norm{\dual_i - \dual_j} \ge \hreg\,\bar{r}$ \Comment{\eqref{eq:appendix-blocks-off}, \eqref{eq:appendix-blocks-diag}}
    \State $\mathrm{B}_i \gets \Nmat_{\shift_i\shift_i}$ for all $i$ \Comment{before the pair terms}
    \State $R \gets \beta\,\sigma\,\bar{r}$ \Comment{pair radius}
    \ForAll{facets $\face_{ij}$ with $d_{ij} = \norm{\dual_j - \dual_i} < R$}
      \State $\mathbf{e} \gets (\dual_j - \dual_i) / d_{ij}$
      \State $\rho^{\max}_{ij} \gets 2 \max(\mathbf{e}\tr\mathrm{B}_i\,\mathbf{e},\, \mathbf{e}\tr\mathrm{B}_j\,\mathbf{e})$
      \State $\rho_{ij} \gets \rho^{\max}_{ij}\,(1 - d_{ij}/R)^4$ \Comment{\eqref{eq:results-pair-reg}}
      \State $\Nmat_{\shift_i\shift_i} \mathrel{+}= \rho_{ij}\,\mathbf{e}\mathbf{e}\tr$, \ $\Nmat_{\shift_j\shift_j} \mathrel{+}= \rho_{ij}\,\mathbf{e}\mathbf{e}\tr$
    \EndFor
    \State $\Nmat_{\shift_i\shift_i} \mathrel{+}= \hreg\,\pvol_i\,r_i^2\,\mathrm{I}$ for all $i$, with $r_i = \pvol_i^{1/d}$ \Comment{shift regularization}
    \State $\Nmat_{\weight_i\weight_i} \mathrel{+}= \hreg\,\pvol_i$ for all $i$ \Comment{weight regularization}
    \State \Return $\Nmat$
  \end{algorithmic}
\end{algorithm}

Without shifts, as for $\dualenergy$, there is no pair term and only the weight regularization applies.
The callers set the activity $\sigma$ to the ratio of the current shift residual $r_\shift$ to its value $r^0_\shift$ at the first iteration, capped at one, so that the pair term fades out as the solve converges, and the radius ratio to $\beta \!=\! 0.3$.
If Algorithm~\ref{alg:newton} or~\ref{alg:flow-constrained-newton} stops without converging, after $100$ iterations or when Algorithm~\ref{alg:lm-step} rejects $60$ trials in a row, it restarts from its initial multipliers with $\beta$ doubled, up to three times.
When the predicted increase in Algorithm~\ref{alg:lm-step} is below the round-off of $\jointdual_\alpha$, a trial is accepted instead if it decreases the residual without decreasing $\jointdual_\alpha$ beyond round-off.

\begin{algorithm}[H]
  \caption{Newton solve of $\dualenergy$.}
  \label{alg:volume}
  \begin{algorithmic}[1]
    \Require \algwrap{sites $\qq$, initial weights $\weight^0$, volumes $\pvol$, tolerance $\tau_\weight$}
    \State $\weight \gets \weight^0$
    \State build the power diagram of the sites $\qq$ with weights $\weight$
    \Loop
      \State $\mathbf{g} \gets \nabla\dualenergy(\weight)$ \Comment{\eqref{eq:volume-dual}}
      \If{$\max_i \lvert g_i \rvert / \pvol_i \le \tau_\weight$} \Comment{\eqref{eq:results-termination}}
        \State \Return $\weight$
      \EndIf
      \State $\Nmat \gets$ Alg.~\ref{alg:hessian-reg} at $\weight$ with $\sigma = 0$
      \State solve $\Nmat\,\delta\weight = \mathbf{g}$
      \State $\weight \gets$ Alg.~\ref{alg:line-search} with $\weight$ and $\delta\weight$
    \EndLoop
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
  \caption{Backtracking line search on $\dualenergy$.}
  \label{alg:line-search}
  \begin{algorithmic}[1]
    \Require \algwrap{sites $\qq$, weights $\weight$, step $\delta\weight$, volumes $\pvol$}
    \State $\omega \gets 1$
    \Loop
      \State $\weight' \gets \weight + \omega\,\delta\weight$
      \State build the power diagram of the sites $\qq$ with weights $\weight'$
      \If{no cell is empty and $\dualenergy(\weight') > \dualenergy(\weight)$}
        \State \Return $\weight'$ and its power diagram
      \EndIf
      \State $\omega \gets \omega / 2$
    \EndLoop
  \end{algorithmic}
\end{algorithm}
